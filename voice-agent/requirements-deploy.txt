# Container / Cloud Shell requirements. Deliberately NOT requirements.txt.
#
# The webrtc extra pulls aiortc, PyAV and opencv — roughly 300 MB of native
# wheels — for a transport Cloud Run cannot serve anyway (WebRTC media is UDP,
# Cloud Run takes none). Leaving it out makes the build faster and removes the
# largest source of native-compilation failures.
#
# Consequence: /web/offer returns an error in this image. /web/ws, which is
# what the widget uses by default, is unaffected. Use requirements.txt on a VM
# if you want WebRTC (see deploy/gce.sh).
#
# Versions below are the ones pip actually resolved against pipecat 1.7.0, not
# hand-picked. An earlier hand-picked pydantic pin was incompatible with
# pipecat and broke every install — pin what resolves, or don't pin.
pipecat-ai[sarvam,silero]==1.7.0

fastapi==0.141.1
uvicorn[standard]==0.52.1
loguru==0.7.3

plivo==4.62.0
google-cloud-firestore==2.28.1
